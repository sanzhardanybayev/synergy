- [ ] **Step 6: Run, verify fail; implement carry-forward in reconcile.ts**

Logic: a file's insight is carried forward iff every review item of that path in the NEW snapshot is a carried-forward match of a previous item of the same path, and the file had an insight before.

```ts
function carryForwardFileInsights(
  previousInsights: ReviewInsights,
  previousSnapshot: ReviewSnapshot,
  nextSnapshot: ReviewSnapshot,
  carriedItemIds: ReadonlySet<string>, // next-revision item ids that were carried forward
): ReviewFileInsight[] | undefined {
  const previous = previousInsights.files;
  if (!previous || previous.length === 0) return undefined;
  const nextItemsByPath = new Map<string, ReviewItem[]>();
  for (const item of nextSnapshot.items) {
    const list = nextItemsByPath.get(item.path) ?? [];
    list.push(item);
    nextItemsByPath.set(item.path, list);
  }
  const carried = previous.filter((file) => {
    const items = nextItemsByPath.get(file.path);
    return items !== undefined && items.every((item) => carriedItemIds.has(item.id));
  });
  return carried.length > 0 ? carried : undefined;
}
```

Wire it where reconcile builds the next revision's insights (it currently produces empty/pending insights for changed items; integrate with that flow - read the existing code and place file carry-forward alongside item carry-forward).

```ts
import { createReviewStore } from '@synergy/review-core';

function subjectLabel(source: ReviewSource): string {
  switch (source.kind) {
    case 'pr': return `PR #${source.number}`;
    case 'staged': return 'Staged changes';
    case 'unstaged': return 'Unstaged changes';
    case 'scope': return `Scope: ${source.patterns.join(', ')}`;
  }
}

export async function buildReviewIndex(projectRoot: string): Promise<ReviewIndexResponse> {
  const store = createReviewStore(projectRoot);
  let workspaces: ReviewWorkspace[];
  try { workspaces = store.listWorkspaces(); } catch { return { reviews: [] }; }
  const reviews = workspaces.map((ws) => {
    try {
      const bundle = store.readBundle(ws.id, ws.currentRevisionId);
      const items = bundle.snapshot.items;
      const reviewed = items.filter((item) => {
        const s = bundle.progress.items[item.id]?.status;
        return s === 'reviewed' || s === 'carried-forward';
      }).length;
      const openQuestions = bundle.questions.filter(
        (q) => q.status === 'queued' || q.status === 'processing',
      ).length;
      return {
        workspaceId: ws.id, revisionId: ws.currentRevisionId,
        subject: subjectLabel(ws.source), sourceKind: ws.source.kind,
        itemCount: items.length, reviewedCount: reviewed,
        openQuestions, updatedAt: ws.updatedAt,
      };
    } catch (err) {
      return {
        workspaceId: ws.id, revisionId: ws.currentRevisionId,
        subject: ws.id, sourceKind: ws.source.kind,
        itemCount: 0, reviewedCount: 0, openQuestions: 0,
        updatedAt: ws.updatedAt, degraded: String(err),
      };
    }
  });
  reviews.sort((a, b) => b.updatedAt.localeCompare(a.updatedAt));
  return { reviews };
}
```
